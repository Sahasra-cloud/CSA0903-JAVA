public class NthLargest {
    public static void main(String args[]) {

        int a[]={14,67,48,23,5,62};
        int temp;

        for(int i=0;i<a.length-1;i++){
            for(int j=i+1;j<a.length;j++){
                if(a[i]>a[j]){
                    temp=a[i];
                    a[i]=a[j];
                    a[j]=temp;
                }
            }
        }

        int n=4;

        System.out.println("Nth Largest = "+a[a.length-n]);
    }
}
