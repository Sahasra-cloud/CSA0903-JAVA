public class AverageNumber {
    public static void main(String args[]) {

        int a[] = {7,-2,9,-8,-6,-4,10};

        int ps=0, ns=0;
        int pc=0, nc=0;

        for(int i=0;i<a.length;i++){

            if(a[i]>=0){
                ps = ps + a[i];
                pc++;
            }
            else{
                ns = ns + a[i];
                nc++;
            }
        }

        System.out.println("Positive Average = " + (double)ps/pc);
        System.out.println("Negative Average = " + (double)ns/nc);
    }
}
