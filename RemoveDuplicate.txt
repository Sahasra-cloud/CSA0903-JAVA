public class RemoveDuplicates {
    public static void main(String args[]) {

        int a[]={10,20,20,30,40,40,50};

        for(int i=0;i<a.length;i++){

            int flag=0;

            for(int j=0;j<i;j++){

                if(a[i]==a[j]){
                    flag=1;
                    break;
                }
            }

            if(flag==0)
                System.out.print(a[i]+" ");
        }
    }
}
